%%%---------------------------------------------------------------------
%%% @doc PortChannel manager.
%%%
%%%      Runs on every node, takes decisions on the master only. Keeps a
%%%      derived cache of the groups, offers the query and facade API for
%%%      CLI, WUI and services, and reconciles the configuration after a
%%%      configuration load, a master change or a failed transaction.
%%%
%%%      The manager never synchronises settings by itself: that is the
%%%      planner's job inside the transaction. Everything here is either
%%%      a query, a facade over dummy_conf or an idempotent repair.
%%% @end
%%%---------------------------------------------------------------------

-module (portchannel_mgr).
-behaviour (gen_server).

-include ("config.hrl").
-include ("validator.hrl").
-include ("portchannel.hrl").
-include ("debug.hrl").

%% Runtime switch and load mode
-export ([enabled/0,
          load_in_progress/0]).

%% Queries
-export ([exists/1,
          members/1,
          pc_of/1,
          portchannels/0,
          pc_config/1,
          status/0]).

%% Facade
-export ([create/2,
          delete/1,
          join/3,
          leave/1]).

%% Notifications
-export ([load_started/0,
          load_finished/0,
          transaction_failed/1,
          committed/1,
          stack_changed/0,
          audit/0,
          audit/1]).

-export ([start_link/0]).

-export ([init/1,
          handle_call/3,
          handle_cast/2,
          handle_info/2,
          terminate/2,
          code_change/3]).

-record (s, {master = false}).

%%%=====================================================================
%%% Switch
%%%=====================================================================

%% @doc Transitional switch (one release): with the manager disabled the
%%      old per-service synchronisation stays in charge.
-spec enabled () -> boolean ().

enabled () ->
  case application:get_env (manager, ?PORTCHANNEL_MGR_ENV) of
    {ok, Value} when is_boolean (Value) -> Value;
    _                                   -> ?PORTCHANNEL_MGR_DEF
  end.

-spec load_in_progress () -> boolean ().

load_in_progress () ->
  flag (load_in_progress, false).

%%%=====================================================================
%%% Queries (pending view: the configuration as the running transaction
%%% leaves it)
%%%=====================================================================

-spec exists (N :: pos_integer ()) -> boolean ().

exists (N) ->
  case dummy_conf:dget_pending (?PORTCHANNEL_KEY (N)) of
    {ok, _} -> true;
    _       -> false
  end.

-spec members (N :: pos_integer ()) -> [term ()].

members (N) ->
  lists:usort (
    dummy_conf:select_pending (
      [{{?LACP_PORT_CHANNEL_GROUP_KEY ('$1'), {N, '_'}}, [], ['$1']}])).

-spec pc_of (If :: term ()) -> {ok, pos_integer ()} | none.

pc_of (If) ->
  case dummy_conf:dget_pending (?LACP_PORT_CHANNEL_GROUP_KEY (If)) of
    {ok, {N, _Mode}} when is_integer (N) -> {ok, N};
    _                                    -> none
  end.

-spec portchannels () -> [pos_integer ()].

portchannels () ->
  Keys = dummy_conf:match_key_pending (?PORTCHANNEL_KEY ('_')),
  lists:usort ([ N || {?PORTCHANNEL_KEY (N), _V} <- Keys ]).

-spec pc_config (N :: pos_integer ()) -> [{term (), term ()}].

pc_config (N) ->
  PC = {portchannel, N},
  [ {Key, Value}
    || {Key, Value} <- all_pending (),
       case portchannel_registry:classify (Key) of
         {ok, _Class, Vif} -> Vif == PC;
         unknown           -> false
       end ].

-spec status () -> [{pos_integer (), [term ()], [term ()]}].

status () ->
  [ {N, members (N), pc_status (N)} || N <- portchannels () ].

%%%=====================================================================
%%% Facade (runs in the caller process)
%%%=====================================================================

-spec create (N :: pos_integer (), Origin :: manual | auto) ->
        ok | {error, term ()}.

create (N, _Origin) when is_integer (N) ->
  case valid_id (N) of
    true  -> dummy_conf:set (?PORTCHANNEL_KEY (N), true);
    false -> {error, invalid_portchannel}
  end;
create (_N, _Origin) ->
  {error, invalid_portchannel}.

-spec delete (N :: pos_integer ()) -> ok | {error, term ()}.

delete (N) when is_integer (N) ->
  case exists (N) of
    true  -> dummy_conf:unset (?PORTCHANNEL_KEY (N));
    false -> {error, portchannel_not_found}
  end;
delete (_N) ->
  {error, invalid_portchannel}.

-spec join (If :: term (), N :: pos_integer (), Mode :: atom ()) ->
        ok | {error, term ()}.

join (If, N, Mode) ->
  dummy_conf:set (?LACP_PORT_CHANNEL_GROUP_KEY (If), {N, Mode}).

-spec leave (If :: term ()) -> ok | {error, term ()}.

leave (If) ->
  dummy_conf:unset (?LACP_PORT_CHANNEL_GROUP_KEY (If)).

%%%=====================================================================
%%% Notifications
%%%=====================================================================

load_started () ->
  set_flag (load_in_progress, true),
  ok.

load_finished () ->
  set_flag (load_in_progress, false),
  cast ({audit, load_finished}),
  ok.

transaction_failed (Bundle) ->
  cast ({transaction_failed, Bundle}).

committed (Changes) ->
  cast ({committed, Changes}).

stack_changed () ->
  cast (stack_changed).

audit () ->
  audit (manual).

audit (Reason) ->
  cast ({audit, Reason}).

%%%=====================================================================
%%% gen_server
%%%=====================================================================

start_link () ->
  gen_server:start_link ({local, ?MODULE}, ?MODULE, [], []).

init ([]) ->
  tables (),
  dummy_conf_event:add_handler (portchannel_eh),
  stack_event:add_handler (portchannel_eh),
  State = #s{master = is_master ()},
  rebuild_cache (),
  {ok, State}.

handle_call (status, _From, State) ->
  {reply, status (), State};
handle_call (_Request, _From, State) ->
  {reply, {error, undefined}, State}.

handle_cast ({committed, Changes}, State) ->
  ?DBG("[MGR] committed: ~p structural changes~n",
       [length([C || C <- Changes, structural(C)])]),
  case lists:any (fun structural/1, Changes) of
    true ->
      rebuild_cache (),
      notify_changed (Changes);
    false ->
      ok
  end,
  {noreply, State};

handle_cast ({transaction_failed, _Bundle}, State) ->
  %% the configuration is unchanged, the hardware may be half way
  %% through: drive it back to what the configuration says
  do_audit (transaction_failed),
  {noreply, State};

handle_cast ({audit, Reason}, State) ->
  do_audit (Reason),
  {noreply, State};

handle_cast (stack_changed, State) ->
  Master = is_master (),
  NewState = State#s{master = Master},
  case Master andalso not State#s.master of
    true ->
      rebuild_cache (),
      do_audit (master_changed),
      redrive_membership ();
    false ->
      rebuild_cache ()
  end,
  {noreply, NewState};

handle_cast (_Msg, State) ->
  {noreply, State}.

handle_info ({gen_event_EXIT, portchannel_eh, _Reason}, State) ->
  ?DBG ("portchannel_mgr: event handler restarted~n", []),
  catch dummy_conf_event:add_handler (portchannel_eh),
  catch stack_event:add_handler (portchannel_eh),
  {noreply, State};
handle_info (_Info, State) ->
  {noreply, State}.

terminate (_Reason, _State) ->
  ok.

code_change (_OldVsn, State, _Extra) ->
  {ok, State}.

%%%=====================================================================
%%% Cache
%%%=====================================================================

tables () ->
  [ create_table (T) || T <- [?PC_TAB_PC, ?PC_TAB_MEMBER,
                              ?PC_TAB_STATUS, ?PC_TAB_FLAGS] ],
  ok.

create_table (Name) ->
  case ets:info (Name) of
    undefined -> ets:new (Name, [named_table, public, set]);
    _         -> Name
  end.

rebuild_cache () ->
  case ets:info (?PC_TAB_PC) of
    undefined ->
      ok;
    _ ->
      ets:delete_all_objects (?PC_TAB_PC),
      ets:delete_all_objects (?PC_TAB_MEMBER),
      [ ets:insert (?PC_TAB_PC, {N, members (N)}) || N <- portchannels () ],
      [ ets:insert (?PC_TAB_MEMBER, {If, N})
        || N <- portchannels (), If <- members (N) ],
      ok
  end.

pc_status (N) ->
  case ets:info (?PC_TAB_STATUS) of
    undefined -> [];
    _         -> [ {Target, Class, Status}
                   || {{Pc, Target, Class}, Status} <-
                        ets:tab2list (?PC_TAB_STATUS), Pc == N ]
  end.

set_status (N, Target, Class, Status) ->
  case ets:info (?PC_TAB_STATUS) of
    undefined -> ok;
    _         -> ets:insert (?PC_TAB_STATUS,
                             {{N, Target, Class}, {Status, os:timestamp ()}})
  end.

flag (Name, Default) ->
  case ets:info (?PC_TAB_FLAGS) of
    undefined ->
      Default;
    _ ->
      case ets:lookup (?PC_TAB_FLAGS, Name) of
        [{_Name, Value}] -> Value;
        []               -> Default
      end
  end.

set_flag (Name, Value) ->
  create_table (?PC_TAB_FLAGS),
  ets:insert (?PC_TAB_FLAGS, {Name, Value}).

structural ({Key, _Action}) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = membership}, _} -> true;
    {ok, #pc_class{scope = entity}, _}     -> true;
    _                                      -> false
  end.

notify_changed (Changes) ->
  PCs = lists:usort (
          [ N || {Key, _A} <- Changes,
                 N <- [pc_of_key (Key)],
                 N =/= none ]),
  [ catch evm:notify ({portchannel, {changed, N}}) || N <- PCs ],
  ok.

pc_of_key (Key) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = membership}, If} ->
      case pc_of (If) of
        {ok, N} -> N;
        none    -> none
      end;
    {ok, _Class, {portchannel, N}} ->
      N;
    _ ->
      none
  end.

%%%=====================================================================
%%% Audit (reconciliation)
%%%=====================================================================

do_audit (Reason) ->
  case enabled () andalso is_master () andalso not load_in_progress () of
    false -> ok;
    true ->
      ?DBG("[MGR] === AUDIT START (~p) ===~n", [Reason]),
      Migrate = (Reason == load_finished),
      {Sets, Unsets} = audit_config (Migrate),
      ?DBG("[MGR] AUDIT config repair: ~p sets, ~p unsets~n", [length(Sets), length(Unsets)]),
      apply_repair (Sets, Unsets),
      audit_runtime (Reason),
      ok
  end.

%% A1 .. A3, A7: configuration level invariants.
audit_config (Migrate) ->
  All = all_applied (),
  Existing = portchannels (),
  Missing = audit_entities (All, Existing),
  {CopySets, CopyUnsets} = audit_copies (Existing),
  Orphans = audit_orphan_refs (All, Existing),
  Dups = case Migrate of
           true  -> audit_duplicate_refs (All);
           false -> []
         end,
  { Missing ++ CopySets, CopyUnsets ++ Orphans ++ Dups }.

%% A1: every referenced group has an entity key.
audit_entities (All, Existing) ->
  Referenced =
    lists:usort (
      [ N || {Key, Value} <- All,
             N <- [referenced_pc (Key, Value)],
             N =/= none ]),
  [ {?PORTCHANNEL_KEY (N), true}
    || N <- Referenced, not lists:member (N, Existing) ].

referenced_pc (Key, Value) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = membership}, _If} ->
      case Value of
        {N, _Mode} when is_integer (N) -> N;
        _                              -> none
      end;
    {ok, #pc_class{scope = entity}, _} ->
      none;
    {ok, _Class, {portchannel, N}} ->
      N;
    _ ->
      none
  end.

%% A2: a member carries exactly the inherited settings of its group.
audit_copies (Existing) ->
  lists:foldl (
    fun (N, {Sets, Unsets}) ->
      PcKeys = [ {K, V} || {K, V} <- pc_config (N), inherited_key (K) ],
      lists:foldl (
        fun (If, {S0, U0}) ->
          S1 = S0 ++ [ {PortKey, V}
                       || {K, V} <- PcKeys,
                          {ok, PortKey} <- [portchannel_registry:port_key (K, If)],
                          dummy_conf:dget (PortKey) =/= {ok, V} ],
          U1 = U0 ++ [ PortKey
                       || {PortKey, _PV} <- member_inherited_keys (If),
                          not has_pc_counterpart (PortKey, N) ],
          {S1, U1}
        end,
        {Sets, Unsets},
        members (N))
    end,
    {[], []},
    Existing).

inherited_key (Key) ->
  case portchannel_registry:classify (Key) of
    {ok, Class, _Vif} -> portchannel_registry:inherited (Class);
    unknown           -> false
  end.

member_inherited_keys (If) ->
  [ {Key, Value} || {Key, Value} <- all_applied (),
                    inherited_key (Key),
                    case portchannel_registry:classify (Key) of
                      {ok, _C, Vif} -> Vif == If;
                      unknown       -> false
                    end ].

has_pc_counterpart (PortKey, N) ->
  case portchannel_registry:pc_key (PortKey, N) of
    {ok, PcKey} -> dummy_conf:dget (PcKey) =/= {error, not_found};
    unknown     -> true
  end.

%% A3: global references to a group that does not exist any more.
audit_orphan_refs (All, Existing) ->
  [ Key || {Key, _V} <- All,
           case portchannel_registry:classify (Key) of
             {ok, #pc_class{scope = global}, {portchannel, N}} ->
               not lists:member (N, Existing);
             _ ->
               false
           end ].

%% A7 (Q9): global references duplicated onto members by the old
%% synchronisation. Removed once, when the configuration is loaded.
audit_duplicate_refs (All) ->
  [ Key
    || {Key, _V} <- All,
       case portchannel_registry:classify (Key) of
         {ok, #pc_class{scope = global, id = Id}, Vif}
           when Id == igmp_mrouter; Id == igmp_static ->
             duplicate_of_pc (Key, Vif);
         _ ->
           false
       end ].

duplicate_of_pc (Key, Vif) ->
  case portchannel_registry:is_pc (Vif) of
    true ->
      false;
    false ->
      case pc_of (Vif) of
        none ->
          false;
        {ok, N} ->
          case portchannel_registry:pc_key (Key, N) of
            {ok, PcKey} -> dummy_conf:dget (PcKey) =/= {error, not_found};
            unknown     -> false
          end
      end
  end.

apply_repair ([], []) ->
  ok;
apply_repair (Sets, Unsets) ->
  ?DBG ("portchannel_mgr: repair ~p set, ~p unset~n",
        [length (Sets), length (Unsets)]),
  UnsetBundle = case lists:usort (Unsets) of
                  []    -> [];
                  Keys  -> [{unset, Keys}]
                end,
  SetBundle = case Sets of
                [] -> [];
                _  -> [{set, Sets}]
              end,
  Bundles = UnsetBundle ++ SetBundle,
  case catch dummy_conf:transaction (Bundles) of
    ok ->
      ok;
    Error ->
      ?DBG ("portchannel_mgr: repair failed: ~p~n", [Error]),
      ok
  end.

%% A4 .. A6: runtime state of the services.
audit_runtime (Reason) ->
  [ begin
      Members = members (N),
      [ set_status (N, If, membership, ok) || If <- Members ],
      case Reason of
        master_changed -> redrive_pc (N, Members);
        _              -> ok
      end,
      touch_refs (N)
    end
    || N <- portchannels () ],
  ok.

redrive_membership () ->
  [ redrive_pc (N, members (N)) || N <- portchannels () ],
  ok.

redrive_pc (N, Members) ->
  [ begin
      case dummy_conf:dget (?LACP_PORT_CHANNEL_GROUP_KEY (If)) of
        {ok, {N, Mode}} ->
          catch node:call (node:master (), trunk_conf, set_port_conf,
                           [If, [{channel_group, {N, Mode}}]]),
          portchannel_hw:on_member_added (N, If, Mode);
        _ ->
          ok
      end
    end
    || If <- Members ],
  ok.

%% A6: re-drive the hardware expansion of global references.
touch_refs (N) ->
  PC = {portchannel, N},
  Keys = [ Key
           || {Key, _V} <- all_applied (),
              case portchannel_registry:classify (Key) of
                {ok, #pc_class{scope = global}, Vif} -> Vif == PC;
                _                                    -> false
              end ],
  case Keys of
    [] -> ok;
    _  -> catch dummy_conf:reapply (Keys), ok
  end.

%%%=====================================================================
%%% Internal
%%%=====================================================================

valid_id (N) ->
  is_integer (N) andalso
  N >= ?LACP_PORT_CHANNEL_GROUP_MIN andalso
  N =< ?LACP_PORT_CHANNEL_GROUP_MAX.

is_master () ->
  case catch node:this () == node:master () of
    true -> true;
    _    -> false
  end.

all_pending () ->
  dummy_conf:select_pending ([{{'$1', '$2'}, [], [{{'$1', '$2'}}]}]).

all_applied () ->
  dummy_conf:select ([{{'$1', '$2'}, [], [{{'$1', '$2'}}]}]).

cast (Msg) ->
  catch gen_server:cast (?MODULE, Msg),
  ok.



%%%---------------------------------------------------------------------
%%% @doc Configurator of the PortChannel subsystem.
%%%
%%%      Applied last, after every other subsystem of the transaction.
%%%      Drives the services that follow membership rather than
%%%      configuration keys (spanning tree, dhcp snooping bindings,
%%%      errdisable). Never writes configuration.
%%% @end
%%%---------------------------------------------------------------------

-module (config_portchannel).

-include ("config.hrl").
-include ("validator.hrl").
-include ("portchannel.hrl").
-include ("debug.hrl").

-export ([apply/1]).

-spec apply (Opts :: [option ()]) -> ok.

apply (Opts) ->
  case portchannel_mgr:enabled () of
    false -> ok;
    true  -> hooks (Opts)
  end.

%%%=====================================================================
%%% Internal functions
%%%=====================================================================

hooks (Opts) ->
  Deleted = [ N || #co{key = ?PORTCHANNEL_KEY (N), action = unset} <- Opts ],
  Deltas = deltas (Opts),
  ?DBG("[CFG_PC] apply hooks: Deleted=~p, Deltas=~p~n", [Deleted, Deltas]),
  [ portchannel_hw:on_member_added (N, If, Mode)
    || {added, If, N, Mode} <- Deltas ],
  [ portchannel_hw:on_member_removed (N, If)
    || {removed, If, N} <- Deltas, not lists:member (N, Deleted) ],
  [ portchannel_hw:on_pc_deleted (N, [ If || {removed, If, M} <- Deltas,
                                             M == N ])
    || N <- Deleted ],
  ok.

%% The applied table still holds the state before the transaction, the
%% option itself carries the state after it.
deltas (Opts) ->
  lists:flatten (
    [ delta (Opt) || Opt <- Opts, is_membership (Opt) ]).

is_membership (#co{key = Key}) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = membership}, _If} -> true;
    _                                        -> false
  end.

delta (#co{key = ?LACP_PORT_CHANNEL_GROUP_KEY (If) = Key} = Opt) ->
  Before = case dummy_conf:dget (Key) of
             {ok, {OldN, _OldM}} when is_integer (OldN) -> OldN;
             _                                          -> none
           end,
  After = case Opt of
            #co{action = set, value = {NewN, NewM}} when is_integer (NewN) ->
              {NewN, NewM};
            _ ->
              none
          end,
  membership_delta (If, Before, After);
delta (_Opt) ->
  [].

membership_delta (_If, Same, {Same, _Mode}) ->
  [];
membership_delta (If, none, {New, Mode}) ->
  [{added, If, New, Mode}];
membership_delta (If, Old, none) when is_integer (Old) ->
  [{removed, If, Old}];
membership_delta (If, Old, {New, Mode}) when is_integer (Old) ->
  [{removed, If, Old}, {added, If, New, Mode}];
membership_delta (_If, _Before, _After) ->
  [].

  {ok, State};
handle_event ({dummy_conf, transaction_done}, #state{keys = Keys} = State) ->
  ?DBG("[EH] transaction_done: forwarding ~p PC keys to mgr~n", [length(Keys)]),
  portchannel_mgr:committed (lists:reverse (Keys)),

    unknown -> {ok, State};
    {ok, Class, _Vif} -> 
      ?DBG("[EH] collected key: ~p ~p (class: ~p)~n", [Action, Key, Class#pc_class.id]),
      {ok, State#state{keys = [{Key, Action} | Keys]}}
  end;

  portchannel_mgr:stack_changed (),
  {ok, State};
handle_event ({stack, {nodedown, _}}, State) ->
  portchannel_mgr:stack_changed (),
  {ok, State};
handle_event (_Event, State) ->
  {ok, State}.

handle_call (_Request, State) ->
  {ok, ok, State}.

handle_info (_Info, State) ->
  {ok, State}.

terminate (_Reason, _State) ->
  ok.

code_change (_OldVsn, State, _Extra) ->
  {ok, State}.

%%%      Two things live here:

%%%         packet processor accepts for a logical trunk interface, which
%%%         of them it repeats for a port that joins later and which ones
%%%         have to be expanded to physical ports by the manager;
%%%
%%%      2. the hooks for services that are driven by membership rather
%%%         than by configuration keys (spanning tree, dhcp snooping
%%%         bindings, errdisable state).
%%%
%%%      Hooks run on the master only and never touch the configuration.
%%% @end

-module (portchannel_hw).

-include ("debug.hrl").

          on_pc_deleted/2]).

%%%=====================================================================
%%% Membership hooks
%%%=====================================================================

-spec on_member_added (N :: pos_integer (), If :: term (), Mode :: atom ()) -> ok.

on_member_added (N, If, Mode) ->
  on_master (
    fun () ->
      call (stp_mgr, mst_set_trunk_ports, [If, N]),
      call (stp_mgr, event,
            [{set, {?LACP_PORT_CHANNEL_GROUP_KEY (If), {N, Mode}}}]),
      multicall (dhcp_snp, binding_del_on_vif, [If]),
      ok
    end).

-spec on_member_removed (N :: pos_integer (), If :: term ()) -> ok.

on_member_removed (_N, If) ->
  on_master (
    fun () ->
      call (stp_mgr, mst_set_trunk_ports, [If, none]),
      call (stp_mgr, event,
            [{set, {?LACP_PORT_CHANNEL_GROUP_KEY (If),
                    ?LACP_PORT_CHANNEL_GROUP_DEF}}]),
      ok
    end).

-spec on_pc_deleted (N :: pos_integer (), Members :: [term ()]) -> ok.

on_pc_deleted (N, Members) ->
  on_master (
    fun () ->
      [ begin
          call (stp_mgr, mst_set_trunk_ports, [If, none]),
          call (stp_mgr, event,
                [{set, {?LACP_PORT_CHANNEL_GROUP_KEY (If),
                        ?LACP_PORT_CHANNEL_GROUP_DEF}}])
        end
        || If <- Members ],
      multicall (dhcp_snp, binding_del_on_vif, [{portchannel, N}]),
      catch errdisable:activate_iface ({portchannel, N}),
      ok
    end).

%%%---------------------------------------------------------------------
%%% @doc PortChannel transaction planner.
%%%
%%%      The single place where synchronisation between a PortChannel and
%%%      its physical members is decided. Runs inside the dummy_conf
%%%      process, between the validator fold and the apply multicall.
%%%
%%%      Pure with respect to the world: reads the two configuration
%%%      tables (applied = state before the transaction, validated =
%%%      state after it), keeps its own overlay of generated options and
%%%      returns them. No gen_server calls, no hardware, no writes.
%%% @end
%%%---------------------------------------------------------------------

-module (portchannel_planner).

-include ("config.hrl").
-include ("validator.hrl").
-include ("portchannel.hrl").
-include ("debug.hrl").

-export ([plan/4]).

-record (ctx, { applied,
                validated,
                load    = false,
                overlay = #{},
                ops     = [] }).

%%%=====================================================================
%%% API
%%%=====================================================================

%% @doc Plan the PortChannel part of a transaction.
%%      Returns the final option list (original options, possibly with
%%      rewritten apply flags, plus generated ones) and the generated
%%      options alone, so that the caller can put them into the
%%      validated table.
-spec plan (ValidOpts :: [#co{}],
            Applied   :: atom (),
            Validated :: atom (),
            LoadMode  :: boolean ()) ->
        {ok, [#co{}], [#co{}]} | {error, term ()}.

plan (ValidOpts, Applied, Validated, LoadMode) ->
  ?DBG("[PLANNER] === plan called: ~p opts, LoadMode=~p ===~n",
       [length(ValidOpts), LoadMode]),
  try
    Ctx0 = #ctx{applied = Applied, validated = Validated, load = LoadMode},
    case is_relevant (ValidOpts, Ctx0) of
      false ->
        ?DBG("[PLANNER] not relevant, skipping~n", []),
        {ok, ValidOpts, []};
      true ->
        Ctx1 = phase_a (ValidOpts, Ctx0),
        ?DBG("[PLANNER] Phase A done. Overlay size: ~p~n", [maps:size(Ctx1#ctx.overlay)]),
        Ctx2 = phase_b (ValidOpts, Ctx1),
        ?DBG("[PLANNER] Phase B done. Overlay size: ~p~n", [maps:size(Ctx2#ctx.overlay)]),
        Ctx3 = phase_c (ValidOpts, Ctx2),
        Generated = lists:reverse (Ctx3#ctx.ops),
        ?DBG("[PLANNER] Phase C done. Generated ~p ops:~n", [length(Generated)]),
        [ ?DBG("[PLANNER]   -> ~p ~p = ~p (apply=~p)~n",
               [O#co.action, O#co.key, O#co.value, O#co.apply]) || O <- Generated ],
        Final = rewrite_apply (ValidOpts, Ctx3) ++ Generated,
        {ok, Final, Generated}
    end
  catch
    throw:{pc_error, Error} ->
      ?DBG("[PLANNER] REJECTED: ~p~n", [Error]),
      {error, Error};
    Class:Reason ->
      ?DBG ("[PLANNER] CRASH: ~w: ~p~n~p~n",
            [Class, Reason, erlang:get_stacktrace ()]),
      {error, ?EPCINTERNAL}
  end.

%%%=====================================================================
%%% Phase A: structure
%%%=====================================================================

phase_a (Opts, Ctx0) ->
  Deleted = [ N || #co{key = Key, action = unset} <- Opts,
                   {ok, N} <- [entity_pc (Key)] ],
  Ctx1 = lists:foldl (fun cascade_delete/2, Ctx0, Deleted),
  Ctx2 = auto_create (Opts, Deleted, Ctx1),
  ok = check_global_refs (Opts, Ctx2),
  ok = check_denied (Opts),
  ok = check_member_edit (Opts, Ctx2),
  Ctx3 = lists:foldl (fun (If, Acc) -> check_join (If, Acc) end,
                      Ctx2,
                      [ If || {added, If, _N} <- membership_delta (Opts, Ctx2) ]),
  Ctx3.

%% Cascade: members leave, inherited copies are reset, PC keys and global
%% references go away (Q7 + Q1).
cascade_delete (N, Ctx0) ->
  PC = {portchannel, N},
  Members = after_members (N, Ctx0),
  Ctx1 =
    lists:foldl (
      fun (If, Acc0) ->
        Acc1 = emit (Acc0, unset, ?LACP_PORT_CHANNEL_GROUP_KEY (If), undefined, true),
        strip_member (If, N, Acc1)
      end,
      Ctx0,
      Members),
  Ctx2 =
    lists:foldl (
      fun ({Key, _V}, Acc) ->
        case portchannel_registry:classify (Key) of
          {ok, Class, _Vif} ->
            emit (Acc, unset, Key, undefined,
                  portchannel_registry:pc_apply (Class, []));
          unknown ->
            emit (Acc, unset, Key, undefined, true)
        end
      end,
      Ctx1,
      vif_keys (PC, Ctx1) ++ refs_to (PC, Ctx1)),
  drop_value_refs (PC, Ctx2).

%% Inherited settings of a member are reset to default when the whole
%% group is removed (Q1).
strip_member (If, N, Ctx0) ->
  lists:foldl (
    fun ({Key, _V}, Acc) ->
      case portchannel_registry:classify (Key) of
        {ok, Class, _Vif} ->
          case portchannel_registry:inherited (Class) of
            true ->
              case portchannel_registry:pc_key (Key, N) of
                {ok, PcKey} ->
                  case after_lookup (PcKey, Acc) of
                    none -> Acc;
                    _    -> emit (Acc, unset, Key, undefined, true)
                  end;
                unknown -> Acc
              end;
            false -> Acc
          end;
        unknown -> Acc
      end
    end,
    Ctx0,
    vif_keys (If, Ctx0)).

%% Keys whose value points at the PortChannel (flex-link backup,
%% protected destination).
drop_value_refs ({portchannel, N} = PC, Ctx0) ->
  Refs = [ Key
           || {Key, Value} <- all_after (Ctx0),
              value_refers_to (Value, PC, N) ],
  lists:foldl (fun (Key, Acc) -> emit (Acc, unset, Key, undefined, true) end,
               Ctx0,
               Refs).

value_refers_to ({portchannel, N}, _PC, N)  -> true;
value_refers_to ([portchannel, N], _PC, N)  -> true;
value_refers_to (_, _, _)                   -> false.

%% Q8: any PortChannel scoped write creates the group, a global reference
%% to a missing group is an error.
auto_create (Opts, Deleted, Ctx0) ->
  Wanted =
    lists:usort (
      [ N || #co{key = Key} = Opt <- Opts,
             {ok, N} <- [creating_pc (Key, Opt)],
             not lists:member (N, Deleted) ]),
  lists:foldl (
    fun (N, Acc) ->
      case after_lookup (?PORTCHANNEL_KEY (N), Acc) of
        none -> emit (Acc, set, ?PORTCHANNEL_KEY (N), true, true);
        _    -> Acc
      end
    end,
    Ctx0,
    Wanted).

creating_pc (Key, Opt = #co{action = Action}) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = global}, _Vif} ->
      %% a reference never creates the group it points at
      unknown;
    {ok, #pc_class{scope = entity}, {portchannel, N}} when Action == set ->
      {ok, N};
    {ok, #pc_class{scope = membership}, _If} ->
      %% channel-group N: the group is named by the value
      case Opt of
        #co{action = set, value = {N, _Mode}} when is_integer (N) -> {ok, N};
        _                                                         -> unknown
      end;
    {ok, _Class, {portchannel, N}} when Action == set ->
      {ok, N};
    _ ->
      unknown
  end.

check_global_refs (Opts, Ctx) ->
  lists:foreach (
    fun (#co{key = Key, action = set}) ->
          case portchannel_registry:classify (Key) of
            {ok, #pc_class{scope = global}, {portchannel, N}} ->
              case after_lookup (?PORTCHANNEL_KEY (N), Ctx) of
                none -> fail (?EPCNOTFOUND);
                _    -> ok
              end;
            _ ->
              ok
          end;
        (_) ->
          ok
    end,
    Opts),
  ok.

check_denied (Opts) ->
  lists:foreach (
    fun (#co{key = Key}) ->
      case portchannel_registry:classify (Key) of
        {ok, Class, {portchannel, _N}} ->
          case portchannel_registry:applicable (Class) of
            allow        -> ok;
            {deny, Tag}  -> fail ({error, {Tag, atom_to_list (Tag)}})
          end;
        _ ->
          ok
      end
    end,
    Opts),
  ok.

%% Q5: an inherited setting may not be edited on a member directly.
check_member_edit (_Opts, #ctx{load = true}) ->
  ok;
check_member_edit (Opts, Ctx) ->
  lists:foreach (
    fun (#co{key = Key} = Opt) ->
      case portchannel_registry:classify (Key) of
        {ok, Class, Vif} ->
          case portchannel_registry:is_pc (Vif) of
            true ->
              ok;
            false ->
              case portchannel_registry:inherited (Class) of
                true  -> check_member_key (Opt, Class, Vif, Ctx);
                false -> ok
              end
          end;
        unknown ->
          ok
      end
    end,
    Opts),
  ok.

check_member_key (#co{key = Key} = Opt, _Class, If, Ctx) ->
  case pc_of (If, Ctx) of
    none ->
      ok;
    {ok, N} ->
      case portchannel_registry:pc_key (Key, N) of
        unknown ->
          ok;
        {ok, PcKey} ->
          PcValue = after_lookup (PcKey, Ctx),
          OptValue = case Opt of
                       #co{action = set, value = V} -> {ok, V};
                       #co{action = unset}          -> none
                     end,
          case PcValue == OptValue of
            true  -> ok;
            false -> fail (?EPCINHERITED)
          end
      end
  end.

%% Port type, group capacity and settings that forbid joining.
check_join (If, Ctx) ->
  case pc_of (If, Ctx) of
    none ->
      Ctx;
    {ok, N} ->
      Members = after_members (N, Ctx),
      check_capacity (Members),
      check_port_type (If, Members),
      check_conflicts (If, Ctx),
      Ctx
  end.

check_capacity (Members) ->
  case length (Members) > portchannel_registry:max_members () of
    true  -> fail (?EINVALGROUPSIZE);
    false -> ok
  end.

check_port_type ({Type, _, _}, Members) ->
  Others = [ T || {T, _, _} <- Members, T =/= Type ],
  case Others of
    [] -> ok;
    _  -> fail (?EINVALIDPORTTYPE)
  end;
check_port_type (_If, _Members) ->
  ok.

check_conflicts (If, Ctx) ->
  lists:foreach (
    fun ({KeyFun, Default, Error}) ->
      case after_lookup (KeyFun (If), Ctx) of
        none          -> ok;
        {ok, Default} -> ok;
        {ok, _Other}  -> fail (Error)
      end
    end,
    portchannel_registry:join_conflicts ()),
  ok.

%%%=====================================================================
%%% Phase B: settings closure
%%%=====================================================================

phase_b (Opts, Ctx0) ->
  Delta = membership_delta (Opts, Ctx0),
  Ctx1 = lists:foldl (fun ({added, If, N}, Acc)   -> join_closure (If, N, Acc);
                          ({removed, If, N}, Acc) -> leave_closure (If, N, Acc)
                      end,
                      Ctx0,
                      Delta),
  lists:foldl (fun (Opt, Acc) -> setting_closure (Opt, Acc) end, Ctx1, Opts).

%% A port that joins takes over every inherited setting of the group and
%% loses its own values in those classes (port_to_pc semantics).
join_closure (If, N, Ctx0) ->
  PC = {portchannel, N},
  Ctx1 =
    lists:foldl (
      fun ({PcKey, Value}, Acc) ->
        case portchannel_registry:classify (PcKey) of
          {ok, Class, _} ->
            case portchannel_registry:inherited (Class) of
              true  -> copy_to_member (Class, PcKey, Value, If, Acc);
              false -> Acc
            end;
          unknown ->
            Acc
        end
      end,
      Ctx0,
      vif_keys (PC, Ctx0)),
  %% settings the port had on its own in inherited classes are dropped
  lists:foldl (
    fun ({PortKey, _V}, Acc) ->
      case portchannel_registry:classify (PortKey) of
        {ok, Class, _} ->
          case portchannel_registry:inherited (Class) of
            true ->
              case portchannel_registry:pc_key (PortKey, N) of
                {ok, PcKey} ->
                  case after_lookup (PcKey, Acc) of
                    none -> emit (Acc, unset, PortKey, undefined, true);
                    _    -> Acc
                  end;
                unknown ->
                  Acc
              end;
            false ->
              Acc
          end;
        unknown ->
          Acc
      end
    end,
    Ctx1,
    vif_keys (If, Ctx1)).

copy_to_member (Class, PcKey, Value, If, Ctx) ->
  case portchannel_registry:port_key (PcKey, If) of
    unknown ->
      Ctx;
    {ok, PortKey} ->
      Current = after_lookup (PortKey, Ctx),
      case portchannel_registry:join_action (Class) of
        detach_and_touch ->
          Ctx1 = case Current of
                   none -> Ctx;
                   _    -> emit (Ctx, unset, PortKey, undefined, true)
                 end,
          emit (Ctx1, set, PortKey, Value, false);
        copy_only ->
          case Current of
            {ok, Value} -> Ctx;
            _           -> emit (Ctx, set, PortKey, Value, false)
          end;
        replay_copy ->
          case Current of
            {ok, Value} -> Ctx;
            _           -> emit (Ctx, set, PortKey, Value, true)
          end
      end
  end.

%% Q2: a port that leaves keeps the settings; classes whose member copy
%% was not applied on its own are materialised now.
leave_closure (If, _N, Ctx0) ->
  lists:foldl (
    fun ({PortKey, Value}, Acc) ->
      case portchannel_registry:classify (PortKey) of
        {ok, Class, _} ->
          case portchannel_registry:inherited (Class) andalso
               portchannel_registry:leave_action (Class) == keep_materialize of
            true  -> emit (Acc, set, PortKey, Value, true);
            false -> Acc
          end;
        unknown ->
          Acc
      end
    end,
    Ctx0,
    vif_keys (If, Ctx0)).

%% A setting written on the group is mirrored to the current members.
setting_closure (#co{key = Key} = Opt, Ctx) ->
  case portchannel_registry:classify (Key) of
    {ok, Class, {portchannel, N}} ->
      case portchannel_registry:inherited (Class) of
        true  -> fan_out (Class, Opt, N, Ctx);
        false -> Ctx
      end;
    _ ->
      Ctx
  end.

fan_out (Class, #co{key = PcKey, action = Action, value = Value}, N, Ctx) ->
  Apply = portchannel_registry:copy_apply (Class),
  lists:foldl (
    fun (If, Acc) ->
      case portchannel_registry:port_key (PcKey, If) of
        unknown ->
          Acc;
        {ok, PortKey} ->
          case Action of
            set ->
              case after_lookup (PortKey, Acc) of
                {ok, Value} -> Acc;
                _           -> emit (Acc, set, PortKey, Value, Apply)
              end;
            unset ->
              case after_lookup (PortKey, Acc) of
                none -> Acc;
                _    -> emit (Acc, unset, PortKey, undefined, Apply)
              end
          end
      end
    end,
    Ctx,
    after_members (N, Ctx)).

%%%=====================================================================
%%% Phase C: hardware expansion refresh
%%%=====================================================================

phase_c (Opts, Ctx0) ->
  Changed = lists:usort ([ N || {_What, _If, N} <- membership_delta (Opts, Ctx0) ]),
  lists:foldl (fun (N, Acc) -> refresh_expansion (N, Acc) end, Ctx0, Changed).

refresh_expansion (N, Ctx0) ->
  PC = {portchannel, N},
  case after_lookup (?PORTCHANNEL_KEY (N), Ctx0) of
    none ->
      %% the group is gone, nothing to recompute
      Ctx0;
    _ ->
      Members = after_members (N, Ctx0),
      lists:foldl (
        fun ({Key, Value}, Acc) ->
          case portchannel_registry:classify (Key) of
            {ok, Class, _} ->
              case portchannel_registry:touch_on_members_changed (Class) of
                true  -> touch (Class, Key, Value, Members, Acc);
                false -> Acc
              end;
            unknown ->
              Acc
          end
        end,
        Ctx0,
        vif_keys (PC, Ctx0) ++ refs_to (PC, Ctx0))
  end.

touch (Class, Key, Value, Members, Ctx) ->
  case portchannel_registry:pc_apply (Class, Members) of
    true ->
      emit (Ctx, set, Key, Value, true);
    false ->
      %% shared resource of an empty group: release it, keep the config
      Ctx1 = emit (Ctx, unset, Key, undefined, true),
      emit (Ctx1, set, Key, Value, false)
  end.

%%%=====================================================================
%%% Membership
%%%=====================================================================

membership_delta (Opts, Ctx) ->
  lists:foldl (
    fun (#co{key = Key} = Opt, Acc) ->
      case portchannel_registry:classify (Key) of
        {ok, #pc_class{scope = membership}, If} ->
          Before = group_of (before_lookup (Key, Ctx)),
          After  = group_of (case Opt of
                               #co{action = set, value = V} -> {ok, V};
                               #co{action = unset}          -> none
                             end),
          delta (If, Before, After) ++ Acc;
        _ ->
          Acc
      end
    end,
    [],
    Opts).

group_of ({ok, {N, _Mode}}) when is_integer (N) -> N;
group_of (_)                                    -> none.

delta (_If, Same, Same)      -> [];
delta (If, none, New)        -> [{added, If, New}];
delta (If, Old, none)        -> [{removed, If, Old}];
delta (If, Old, New)         -> [{removed, If, Old}, {added, If, New}].

after_members (N, Ctx) ->
  Base = ets:select (Ctx#ctx.validated,
                     [{{?LACP_PORT_CHANNEL_GROUP_KEY ('$1'), {N, '_'}}, [], ['$1']}]),
  Overlay = Ctx#ctx.overlay,
  Added = [ If || {Key, {set, {G, _}}} <- maps:to_list (Overlay),
                  G == N,
                  {ok, #pc_class{scope = membership}, If} <-
                    [portchannel_registry:classify (Key)] ],
  Removed = [ If || {Key, Entry} <- maps:to_list (Overlay),
                    {ok, #pc_class{scope = membership}, If} <-
                      [portchannel_registry:classify (Key)],
                    not_in_group (Entry, N) ],
  lists:usort ((Base ++ Added) -- Removed).

not_in_group (unset, _N)          -> true;
not_in_group ({set, {G, _}}, N)   -> G =/= N;
not_in_group (_, _)               -> true.

pc_of (If, Ctx) ->
  case after_lookup (?LACP_PORT_CHANNEL_GROUP_KEY (If), Ctx) of
    {ok, {N, _Mode}} when is_integer (N) -> {ok, N};
    _                                    -> none
  end.

entity_pc (Key) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = entity}, {portchannel, N}} -> {ok, N};
    _                                                 -> unknown
  end.

%%%=====================================================================
%%% Option bookkeeping
%%%=====================================================================

is_relevant (Opts, Ctx) ->
  lists:any (fun (Opt) -> relevant_opt (Opt, Ctx) end, Opts).

relevant_opt (#co{key = Key}, Ctx) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = membership}, _} ->
      true;
    {ok, #pc_class{scope = entity}, _} ->
      true;
    {ok, Class, Vif} ->
      case portchannel_registry:is_pc (Vif) of
        true ->
          true;
        false ->
          %% a setting written on a port that is a member of a group
          portchannel_registry:inherited (Class) andalso
          pc_of (Vif, Ctx) =/= none
      end;
    unknown ->
      false
  end.

%% The apply flag of a PortChannel key depends on how the class reaches
%% the hardware: shadow classes are stored, not executed.
rewrite_apply (Opts, Ctx) ->
  [ rewrite_opt (Opt, Ctx) || Opt <- Opts ].

rewrite_opt (#co{key = Key} = Opt, Ctx) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = interface} = Class, {portchannel, N}} ->
      Members = after_members (N, Ctx),
      Opt#co{apply = portchannel_registry:pc_apply (Class, Members)};
    _ ->
      Opt
  end.

emit (Ctx, set, Key, Value, Apply) ->
  Op = #co{key = Key, value = Value, action = set, apply = Apply},
  Ctx#ctx{ops = [Op | Ctx#ctx.ops],
          overlay = maps:put (Key, {set, Value}, Ctx#ctx.overlay)};
emit (Ctx, unset, Key, _Value, Apply) ->
  Op = #co{key = Key, action = unset, apply = Apply},
  Ctx#ctx{ops = [Op | Ctx#ctx.ops],
          overlay = maps:put (Key, unset, Ctx#ctx.overlay)}.

fail (Error) ->
  throw ({pc_error, Error}).

%%%=====================================================================
%%% Table access (after = validated + overlay, before = applied)
%%%=====================================================================

after_lookup (Key, #ctx{overlay = Overlay, validated = Tab}) ->
  case maps:find (Key, Overlay) of
    {ok, unset}      -> none;
    {ok, {set, V}}   -> {ok, V};
    error ->
      case ets:lookup (Tab, Key) of
        []          -> none;
        [{_Key, V}] -> {ok, V}
      end
  end.

before_lookup (Key, #ctx{applied = Tab}) ->
  case ets:lookup (Tab, Key) of
    []          -> none;
    [{_Key, V}] -> {ok, V}
  end.

%% Every key of the given VIF, as the transaction leaves it.
vif_keys (Vif, Ctx) ->
  [ {Key, Value} || {Key, Value} <- all_after (Ctx),
                    key_of_vif (Key, Vif) ].

key_of_vif (Key, Vif) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = Scope}, Vif} when Scope == interface ->
      true;
    _ ->
      false
  end.

%% Global references pointing at the given PortChannel.
refs_to (Vif, Ctx) ->
  [ {Key, Value} || {Key, Value} <- all_after (Ctx),
                    ref_of_vif (Key, Vif) ].

ref_of_vif (Key, Vif) ->
  case portchannel_registry:classify (Key) of
    {ok, #pc_class{scope = global}, Vif} -> true;
    _                                    -> false
  end.

all_after (#ctx{validated = Tab, overlay = Overlay}) ->
  Base = ets:foldl (fun ({Key, Value}, Acc) ->
                        case maps:is_key (Key, Overlay) of
                          true  -> Acc;
                          false -> [{Key, Value} | Acc]
                        end
                    end,
                    [],
                    Tab),
  Base ++ [ {Key, Value} || {Key, {set, Value}} <- maps:to_list (Overlay) ].






%%%---------------------------------------------------------------------
%%% @doc PortChannel settings registry.
%%%
%%%      Declarative description of every configuration class the
%%%      PortChannel manager has to reason about: how the VIF is embedded
%%%      into the key, whether the setting is inherited by members, how it
%%%      reaches the hardware and what has to happen on membership change.
%%%
%%%      Pure module: no processes, no ETS, no side effects.
%%% @end
%%%---------------------------------------------------------------------

-module (portchannel_registry).

-compile ({no_auto_import, [is_port/1]}).

-include ("config.hrl").
-include ("validator.hrl").
-include ("portchannel.hrl").

-export ([classify/1,
          class/1,
          vif_of/1,
          rekey/2,
          pc_key/2,
          port_key/2,
          applicable/1,
          inherited/1,
          scope_of/1,
          pc_apply/2,
          copy_apply/1,
          join_action/1,
          leave_action/1,
          delete_action/1,
          touch_on_members_changed/1,
          max_members/0,
          join_conflicts/0,
          is_vif/1,
          is_pc/1,
          is_port/1,
          tag_vif/1]).

%%%=====================================================================
%%% API
%%%=====================================================================

%% @doc Classify a configuration key. Unknown keys are not managed:
%%      the planner leaves them exactly as they are today.
-spec classify (Key :: term ()) ->
        {ok, #pc_class{}, Vif :: term ()} | unknown.

classify (Key) ->
  case describe (Key) of
    unknown ->
      unknown;
    ClassId ->
      Class = class (ClassId),
      case vif_at (Class#pc_class.ref, Key) of
        {ok, Vif} ->
          case is_vif (Vif) of
            true  -> {ok, Class, Vif};
            false -> unknown
          end;
        error ->
          unknown
      end
  end.

%% @doc Extract the VIF a key belongs to.
-spec vif_of (Key :: term ()) -> {ok, term ()} | unknown.

vif_of (Key) ->
  case classify (Key) of
    {ok, _Class, Vif} -> {ok, Vif};
    unknown           -> unknown
  end.

%% @doc Rebuild a key for another VIF (PC -> member or member -> PC).
-spec rekey (Key :: term (), Vif :: term ()) -> {ok, term ()} | unknown.

rekey (Key, Vif) ->
  case describe (Key) of
    unknown ->
      unknown;
    ClassId ->
      Class = class (ClassId),
      set_vif (Class#pc_class.ref, Key, Vif)
  end.

-spec pc_key (Key :: term (), N :: pos_integer ()) -> {ok, term ()} | unknown.

pc_key (Key, N) ->
  rekey (Key, {portchannel, N}).

-spec port_key (Key :: term (), If :: term ()) -> {ok, term ()} | unknown.

port_key (Key, If) ->
  rekey (Key, If).

-spec applicable (#pc_class{}) -> allow | {deny, atom ()}.

applicable (#pc_class{applicable = A}) -> A.

-spec inherited (#pc_class{}) -> boolean ().

inherited (#pc_class{inherit = copy, scope = interface}) -> true;
inherited (_)                                            -> false.

-spec scope_of (#pc_class{}) -> atom ().

scope_of (#pc_class{scope = S}) -> S.

%% @doc Should the PC key itself be handed to its configurator?
%%      Members :: current (pending) member list of the PC.
-spec pc_apply (#pc_class{}, Members :: [term ()]) -> boolean ().

pc_apply (#pc_class{realization = member}, _Members)  -> false;
pc_apply (#pc_class{realization = none}, _Members)    -> false;
pc_apply (#pc_class{realization = shared, empty_pc = shadow}, []) -> false;
pc_apply (_Class, _Members)                           -> true.

%% @doc Should a member copy be handed to its configurator in steady state?
-spec copy_apply (#pc_class{}) -> boolean ().

copy_apply (#pc_class{realization = member})   -> true;
copy_apply (#pc_class{realization = software}) -> true;
copy_apply (_)                                 -> false.

%% @doc What has to happen with a member copy when the port joins.
-spec join_action (#pc_class{}) -> replay_copy | copy_only | detach_and_touch.

join_action (#pc_class{realization = shared})   -> detach_and_touch;
join_action (#pc_class{realization = logical})  -> copy_only;
join_action (_)                                 -> replay_copy.

%% @doc What has to happen when the port leaves (settings are kept, Q2).
-spec leave_action (#pc_class{}) -> keep | keep_materialize.

leave_action (#pc_class{realization = member})   -> keep;
leave_action (#pc_class{realization = software}) -> keep;
leave_action (_)                                 -> keep_materialize.

%% @doc What has to happen when the whole PC is deleted (Q1: reset to default).
-spec delete_action (#pc_class{}) -> strip_copies_and_unset | unset.

delete_action (#pc_class{scope = interface, inherit = copy}) -> strip_copies_and_unset;
delete_action (_)                                            -> unset.

%% @doc Classes whose hardware expansion has to be recomputed when the
%%      member list changes.
-spec touch_on_members_changed (#pc_class{}) -> boolean ().

touch_on_members_changed (#pc_class{scope = global})        -> true;
touch_on_members_changed (#pc_class{realization = shared})  -> true;
touch_on_members_changed (_)                                -> false.

-spec max_members () -> pos_integer ().

max_members () -> ?LACP_PORT_CHANNEL_MAX_MEMBERS.

%% @doc Port settings that forbid joining a group while set to a non
%%      default value. {KeyMaker, DefaultValue, ErrorTerm}.
join_conflicts () ->
  [ {fun (If) -> ?DOT1X_PORT_CONTROL_KEY (If) end,
     'force-authorized', ?ECONFDOT1XLACP},
    {fun (If) -> ?PORT_STORM_CONTROL_KEY (If) end,
     [broadcast, 0], ?ECONFQOSLACP},
    {fun (If) -> ?PORT_SEC_EN_KEY (If) end,
     false, ?ECONFPSLACP},
    {fun (If) -> ?PORT_VOICE_VLAN_KEY (If) end,
     0, ?ECONFVOICEVLANLACP}
  ].

%%%=====================================================================
%%% VIF helpers
%%%=====================================================================

is_vif (Vif) -> is_pc (Vif) orelse is_port (Vif).

is_pc ({portchannel, N}) when is_integer (N) -> true;
is_pc (_)                                    -> false.

is_port ({T, U, N}) when is_atom (T), is_integer (U), is_integer (N) -> true;
is_port (_)                                                          -> false.

%% {port, If} for physical ports, {portchannel, N} for groups.
tag_vif ({portchannel, _N} = PC) -> PC;
tag_vif (If)                     -> {port, If}.

untag_vif ({port, If})            -> If;
untag_vif ({portchannel, _} = PC) -> PC;
untag_vif (Other)                 -> Other.

%%%=====================================================================
%%% VIF extraction / substitution

vif_at ({raw, Pos}, Key) when tuple_size (Key) >= Pos ->
  {ok, element (Pos, Key)};
vif_at ({wrapped, Pos}, Key) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {port, Vif} -> {ok, Vif};
    _           -> error
  end;
vif_at ({tagged, Pos}, Key) when tuple_size (Key) >= Pos ->
  {ok, untag_vif (element (Pos, Key))};
vif_at ({acl, Pos}, Key) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {Tagged, _Dest} -> {ok, untag_vif (Tagged)};
    _               -> error
  end;
vif_at ({mst_inst, Pos}, Key) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {{inst, _I}, {port, Vif}} -> {ok, Vif};
    _                         -> error
  end;
vif_at ({igmp_snoop, Pos}, Key) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {{vlan, _V}, {port, Vif}, _K} -> {ok, Vif};
    _                             -> error
  end;
vif_at ({monitor_src, Pos}, Key) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {source, {port, Vif}} -> {ok, Vif};
    _                     -> error
  end;
vif_at ({mclag_member, Pos}, Key) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {member, Vif} -> {ok, Vif};
    _             -> error
  end;
vif_at (_Ref, _Key) ->
  error.

set_vif ({raw, Pos}, Key, Vif) when tuple_size (Key) >= Pos ->
  {ok, setelement (Pos, Key, Vif)};
set_vif ({wrapped, Pos}, Key, Vif) when tuple_size (Key) >= Pos ->
  {ok, setelement (Pos, Key, {port, Vif})};
set_vif ({tagged, Pos}, Key, Vif) when tuple_size (Key) >= Pos ->
  {ok, setelement (Pos, Key, tag_vif (Vif))};
set_vif ({acl, Pos}, Key, Vif) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {_Tagged, Dest} -> {ok, setelement (Pos, Key, {tag_vif (Vif), Dest})};
    _               -> unknown
  end;
set_vif ({mst_inst, Pos}, Key, Vif) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {{inst, I}, {port, _}} ->
      {ok, setelement (Pos, Key, {{inst, I}, {port, Vif}})};
    _ ->
      unknown
  end;
set_vif ({igmp_snoop, Pos}, Key, Vif) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {{vlan, V}, {port, _}, K} ->
      {ok, setelement (Pos, Key, {{vlan, V}, {port, Vif}, K})};
    _ ->
      unknown
  end;
set_vif ({monitor_src, Pos}, Key, Vif) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {source, {port, _}} ->
      {ok, setelement (Pos, Key, {source, {port, Vif}})};
    _ ->
      unknown
  end;
set_vif ({mclag_member, Pos}, Key, Vif) when tuple_size (Key) >= Pos ->
  case element (Pos, Key) of
    {member, _} -> {ok, setelement (Pos, Key, {member, Vif})};
    _           -> unknown
  end;
set_vif (_Ref, _Key, _Vif) ->
  unknown.




%%%---------------------------------------------------------------------
%%% @doc Validator of the PortChannel subsystem.
%%%
%%%      Owns the entity key only. Every policy decision (inheritance,
%%%      capacity, conflicts, denied classes) is taken by the planner, so
%%%      that there is exactly one place where they live.
%%% @end
%%%---------------------------------------------------------------------

-module (validator_portchannel).

-include ("config.hrl").
-include ("validator.hrl").
-include ("portchannel.hrl").

-export ([validate/1]).

-spec validate (AllOpts :: [option ()]) ->
        {[known_checked_option ()], [check_error ()]}.

validate (Opts) ->
  lists:foldl (fun validate_opt/2, {[], []}, Opts).

validate_opt (Opt, {Valid, Errors}) ->
  case validate_key (Opt) of
    ignore       -> {Valid, Errors};
    {ok, NewOpt} -> {[NewOpt | Valid], Errors};
    {Error, Bad} -> {Valid, [{Error, Bad} | Errors]}
  end.

validate_key (#co{key = ?PORTCHANNEL_KEY (N), action = set, value = true} = Opt) ->
  case utils:in_range (?LACP_PORT_CHANNEL_GROUP_MIN,
                       ?LACP_PORT_CHANNEL_GROUP_MAX,
                       N) of
    true  -> {ok, Opt};
    false -> {?EINCHANNEL, Opt}
  end;
validate_key (#co{key = ?PORTCHANNEL_KEY (_N), action = set} = Opt) ->
  {?EINVAL, Opt};
validate_key (#co{key = ?PORTCHANNEL_KEY (_N), action = unset} = Opt) ->
  {ok, Opt};
validate_key (_Opt) ->
  ignore.
